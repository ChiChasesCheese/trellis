def verify_KYC_submissions(stream):
  required_fields = ["business_name", "business_profile_name", "full_statement_descriptor", "short_statement_descriptor", "url", "product_description"]
  current_submission = {}
  it = iter(stream)
  not_yet_seen_fields = set(required_fields)
  # while stream is ongoing
  while ((line := next(it)) != "EOF"):
    if line == "":
      output = ""
      # check if description field, if exists, has len above 5
      def is_verified():
        description = current_submission.get("full_statement_descriptor", "")
        description_length = len(description)

    else:

      if value:
        current_submission[key] = value
        # remove value from not_yet_seem fields if it exists

If the descriptor's length falls outside that range, the submission is

--------------------------------------------------------------------------------

[Revealed after Part 2 passes.]

One more rule. We've noticed a lot of businesses try to set their descriptor
to something generic like "ONLINE STORE". That's useless for the cardholder

Reject any submission whose `full_statement_descriptor`, after trimming

    RETAIL

These are case-insensitive matches against the whole descriptor -- not
substring matches. "RETAIL THERAPY CO" is fine; "retail" alone is not.

import sys


REQUIRED_FIELDS = [
    "business_name",
    "business_profile_name",
    "full_statement_descriptor",
    "short_statement_descriptor",
    "url",
    "product_description",
]

DESCRIPTOR_BLOCKLIST = {
    "ONLINE STORE",
    "ECOMMERCE",
    "RETAIL",
    "SHOP",
    "GENERAL MERCHANDISE",
}


def parse_submissions(text):
    submissions = []
    for block in text.strip().split("\n\n"):
        record = {}
        for line in block.splitlines():
            if "=" not in line:
                continue
            key, value = line.split("=", 1)
            record[key.strip()] = value
        submissions.append(record)
    return submissions


def has_required_fields(record):
    return all(record.get(f, "").strip() for f in REQUIRED_FIELDS)


def descriptor_length_ok(record):
    desc = record.get("full_statement_descriptor", "")
    return 5 <= len(desc) <= 31


def descriptor_not_blocklisted(record):
    desc = record.get("full_statement_descriptor", "").strip().upper()
    return desc not in DESCRIPTOR_BLOCKLIST


RULES = [has_required_fields, descriptor_length_ok, descriptor_not_blocklisted]


def is_verified(record):
    return all(rule(record) for rule in RULES)


def main():
    submissions = parse_submissions(sys.stdin.read())
    for record in submissions:
        name = record.get("business_name", "").strip()
        status = "VERIFIED" if is_verified(record) else "NOT VERIFIED"
        print(f"{status}: {name}")


if __name__ == "__main__":
    main()
